Extraction script to use:
---
import json
from docx import Document

for batch_num, filename, out_file, expected in [
    (20, 'GreyRadius_SEO_Batch20.docx', 'data/industry-pages-batch20.json', 21),
    (21, 'GreyRadius_SEO_Batch21.docx', 'data/industry-pages-batch21.json', 20),
    (22, 'GreyRadius_SEO_Batch22.docx', 'data/industry-pages-batch22.json', 20),
]:
    doc = Document(filename)
    full_text = '\n'.join([p.text for p in doc.paragraphs])
    # Find the JSON array starting after Section 3 heading
    # The JSON starts with [ and ends with the last ]
    start = full_text.find('[\n  {')
    if start == -1:
        start = full_text.find('[{')
    if start == -1:
        start = full_text.find('[')
    end = full_text.rfind(']') + 1
    json_str = full_text[start:end]
    data = json.loads(json_str)
    assert len(data) == expected, f"Expected {expected} pages, got {len(data)}"
    import os
    os.makedirs('data', exist_ok=True)
    with open(out_file, 'w') as f:
        json.dump(data, f, indent=2)
    print(f"Batch {batch_num}: {len(data)} pages saved to {out_file}")
---
